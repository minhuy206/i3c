\chapter{Định dạng \CommandDescriptor{}, \ResponseDescriptor{} và mục DAT}
\label{app:desc}

Phụ lục này trình bày bố trí trường của các cấu trúc dữ liệu trao đổi giữa phần mềm và phần cứng. \CommandDescriptor{} rộng \CommandDescriptorWidth{}-bit (hai DWORD). \ResponseDescriptor{} rộng \ResponseDescriptorWidth{}-bit~\cite{mipi_i3c_tcri} và mục DAT rộng \DatEntryWidth{}-bit.

\section*{\ResponseDescriptor{} (\ResponseDescriptorWidth{}-bit)}

\begin{longtable}{C{3.2cm} C{2.6cm} L{7.0cm}}
	\toprule
	\textbf{Trường}       & \textbf{Bit} & \textbf{Ý nghĩa}                               \\
	\midrule
	\endhead
	\texttt{err}          & [31:28]      & Mã trạng thái lỗi (xem \cref{tab:err-status}). \\
	\texttt{tid}          & [27:24]      & Mã giao dịch.                                  \\
	(reserved)            & [23:16]      & Luôn đọc về 0.                                 \\
	\texttt{data\_length} & [15:0]       & Số byte đã xử lý.                              \\
	\bottomrule
\end{longtable}

\section*{\ImmediateDataTransfer{} Descriptor (\CommandDescriptorWidth{}-bit)}

Dạng \ImmediateDataTransfer{} nhúng tối đa bốn byte dữ liệu ngay trong \CommandDescriptor{}, không dùng
\TxFifo{}~\cite{mipi_i3c_tcri}.

\begin{longtable}{C{4.5cm} L{8.1cm}}
	\toprule
	\textbf{Trường}                                           & \textbf{Ý nghĩa}                                  \\
	\midrule
	\endhead
	\texttt{attr}                                             & Thuộc tính lệnh = \texttt{ImmediateDataTransfer}. \\
	\texttt{tid}                                              & Mã giao dịch.                                     \\
	\texttt{cmd}                                              & Mã CCC/HDR (nếu có).                              \\
	\texttt{cp}                                               & Cờ cho biết có mã CCC hay không.                  \\
	\texttt{dev\_idx}                                         & Chỉ số mục DAT của \Target{} đích.                \\
	\texttt{dtt}                                              & Số byte dữ liệu hợp lệ (0--4).                    \\
	\texttt{mode}                                             & Tốc độ và chế độ truyền.                          \\
	\texttt{rnw}                                              & Hướng (immediate chỉ ghi).                        \\
	\texttt{wroc}                                             & Yêu cầu ghi \ResponseDescriptor{} khi hoàn tất.   \\
	\texttt{toc}                                              & Yêu cầu kết thúc giao dịch khi hoàn tất.          \\
	\texttt{def\_or\_data\_byte1} \ldots \texttt{data\_byte4} & Bốn byte dữ liệu nhúng.                           \\
	\bottomrule
\end{longtable}

\section*{\RegularTransfer{} Descriptor (\CommandDescriptorWidth{}-bit)}

Dạng \RegularTransfer{} dùng cho giao dịch có dữ liệu nằm trong \TxRxFifo{}~\cite{mipi_i3c_tcri}.

\begin{longtable}{C{3.6cm} L{9.0cm}}
	\toprule
	\textbf{Trường}                         & \textbf{Ý nghĩa}                                                      \\
	\midrule
	\endhead
	\texttt{attr}                           & Thuộc tính lệnh = \texttt{RegularTransfer}.                           \\
	\texttt{tid}, \texttt{cmd}, \texttt{cp} & Mã giao dịch, mã CCC và cờ cho biết có mã CCC.                        \\
	\texttt{dev\_idx}                       & Chỉ số mục DAT của \Target{} đích.                                    \\
	\texttt{mode}, \texttt{rnw}             & Chế độ truyền và hướng.                                               \\
	\texttt{sre}                            & Báo \texttt{I3cShortReadErr} khi giao dịch đọc kết thúc sớm.          \\
	\texttt{dbp}                            & Cờ cho biết có \DefiningByte{}.                                       \\
	\texttt{wroc}, \texttt{toc}             & Yêu cầu ghi \ResponseDescriptor{} và kết thúc giao dịch khi hoàn tất. \\
	\texttt{def\_byte}                      & \DefiningByte{} (nếu có).                                             \\
	\texttt{data\_length}                   & Độ dài dữ liệu (byte).                                                \\
	\bottomrule
\end{longtable}

\section*{\AddressAssignment{} Descriptor (\CommandDescriptorWidth{}-bit)}

Dạng \AddressAssignment{} khởi tạo quy trình ENTDAA; dữ liệu \gls{pid}/\gls{bcr}/\gls{dcr}
được thu trực tiếp từ \Target{} trên bus nên \CommandDescriptor{} không mang byte dữ liệu~\cite{mipi_i3c_tcri}.

\begin{longtable}{C{3.6cm} L{9.0cm}}
	\toprule
	\textbf{Trường}     & \textbf{Ý nghĩa}                                                                                                                \\
	\midrule
	\endhead
	\texttt{attr}       & Thuộc tính lệnh = \texttt{AddressAssignment}.                                                                                   \\
	\texttt{tid}        & Mã giao dịch.                                                                                                                   \\
	\texttt{cmd}        & Mã CCC cấp phát địa chỉ (ENTDAA).                                                                                               \\
	\texttt{dev\_idx}   & Chỉ số mục DAT đầu tiên được cấp địa chỉ.                                                                                       \\
	\texttt{dev\_count} & Số \Target{} cần cấp địa chỉ trong một lần chạy; \texttt{dev\_idx}+\texttt{dev\_count} phải nằm trong \DatEntryCount{} mục DAT. \\
	\texttt{wroc}       & Yêu cầu ghi \ResponseDescriptor{} khi hoàn tất.                                                                                 \\
	\texttt{toc}        & Yêu cầu kết thúc giao dịch khi hoàn tất.                                                                                        \\
	\bottomrule
\end{longtable}
