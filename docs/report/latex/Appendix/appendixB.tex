\chapter{Định dạng Command, Response và DAT descriptor}
\label{app:desc}

Phụ lục này trình bày bố trí trường của các cấu trúc dữ liệu trao đổi giữa phần mềm và phần cứng, định
nghĩa trong \texttt{i3c\_pkg} và \texttt{controller\_pkg}, bổ trợ cho \cref{sec:arch-desc}. \CommandDescriptor{} rộng \CommandDescriptorWidth{}-bit (hai DWORD); \ResponseDescriptor{} và mục DAT rộng \ResponseDescriptorWidth{}-bit.

\section*{\ResponseDescriptor{} (\ResponseDescriptorWidth{}-bit)}

\begin{longtable}{C{3.2cm} C{2.6cm} L{7.0cm}}
  \toprule
  \textbf{Trường} & \textbf{Bit} & \textbf{Ý nghĩa} \\
  \midrule
  \endhead
  \texttt{err\_status}  & [31:28] & Mã trạng thái lỗi (xem \cref{tab:err-status}). \\
  \texttt{tid}          & [27:24] & Transaction ID. \\
  (dự trữ)              & [23:16] & Reserved. \\
  \texttt{data\_length} & [15:0]  & Độ dài dữ liệu thực tế. \\
  \bottomrule
\end{longtable}

\section*{Immediate Data Transfer Descriptor (\CommandDescriptorWidth{}-bit)}

Dạng immediate nhúng tối đa bốn byte dữ liệu ngay trong descriptor, tránh phải dùng TX Buffer.

\begin{longtable}{C{3.6cm} L{9.0cm}}
  \toprule
  \textbf{Trường} & \textbf{Ý nghĩa} \\
  \midrule
  \endhead
  \texttt{attr}        & Thuộc tính lệnh = \texttt{ImmediateDataTransfer}. \\
  \texttt{tid}         & Transaction ID. \\
  \texttt{cmd}         & Mã CCC / HDR (nếu có). \\
  \texttt{cp}          & Command present (có CCC hay không). \\
  \texttt{dev\_idx}    & Chỉ số mục DAT của \Target{} đích. \\
  \texttt{dtt}         & Số byte dữ liệu hợp lệ (0--4). \\
  \texttt{mode}        & Tốc độ/chế độ truyền (\texttt{sdr0}\ldots). \\
  \texttt{rnw}         & Hướng (immediate chỉ ghi). \\
  \texttt{wroc}        & Response-on-completion. \\
  \texttt{toc}         & Terminate-on-completion. \\
  \texttt{def\_or\_data\_byte1} \ldots \texttt{data\_byte4} & Bốn byte dữ liệu nhúng. \\
  \bottomrule
\end{longtable}

\section*{Regular Transfer Descriptor (\CommandDescriptorWidth{}-bit)}

Dạng regular dùng cho giao dịch có dữ liệu nằm trong TX/RX Buffer.

\begin{longtable}{C{3.6cm} L{9.0cm}}
  \toprule
  \textbf{Trường} & \textbf{Ý nghĩa} \\
  \midrule
  \endhead
  \texttt{attr}        & Thuộc tính lệnh = \texttt{RegularTransfer}. \\
  \texttt{tid}, \texttt{cmd}, \texttt{cp} & Transaction ID, mã CCC, cờ command present. \\
  \texttt{dev\_idx}    & Chỉ số mục DAT của \Target{} đích. \\
  \texttt{mode}, \texttt{rnw} & Chế độ truyền và hướng. \\
  \texttt{sre}         & Báo đọc kết thúc sớm là ShortReadErr. \\
  \texttt{dbp}         & Defining byte present. \\
  \texttt{wroc}, \texttt{toc} & Response/terminate-on-completion. \\
  \texttt{def\_byte}   & Defining byte (nếu có). \\
  \texttt{data\_length} & Độ dài dữ liệu (byte). \\
  \bottomrule
\end{longtable}

\noindent Ngoài ra còn có Combo Transfer Descriptor (\texttt{combo\_trans\_desc\_t}) và Address Assignment
Descriptor (\texttt{addr\_assign\_desc\_t}, dùng cho ENTDAA với trường \texttt{dev\_count} và
\texttt{dev\_idx}). Mục DAT (\texttt{dat\_entry\_t}) đã mô tả ở \cref{app:csr}.
