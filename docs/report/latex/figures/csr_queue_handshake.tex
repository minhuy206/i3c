% Biểu đồ định thời handshake \texttt{ready}/\texttt{valid} trên đường ghi từ CSR đến Queue.
% Yêu cầu (khai báo trong report.tex): \usepackage{tikz-timing} và các thư viện
% tikz calc, arrows.meta, backgrounds, patterns. Đặt trong figure có \centering
% (không dùng adjustbox: bề rộng tự nhiên đã nhỏ hơn \textwidth).
% Nhãn/chú thích trong hình để tiếng Anh theo yêu cầu.
\definecolor{cAcc}{rgb}{0.10,0.10,0.65}   % CSR accepts write
\definecolor{cXfer}{rgb}{0.10,0.45,0.10}  % FIFO writes D
\definecolor{cRej}{rgb}{0.70,0.10,0.10}   % new write blocked
% \evmark{<node-x>}{<màu>}{<số>}: vạch mốc dọc đứt nét + vòng tròn đánh số,
% đặt phía TRÊN waveform (anchor=south + yshift) để không đè lên tín hiệu.
% TOP/BOT là hai toạ độ định nghĩa trong \extracode của mỗi bảng.
\newcommand{\evmark}[3]{%
  \draw[#2, dashed, line width=0.6pt] (#1 |- TOP) -- (#1 |- BOT);
  \node[circle, fill=#2, text=white, inner sep=0pt, minimum size=8pt,
        font=\tiny\bfseries, anchor=south, yshift=6pt] at (#1 |- TOP) {#3};}

\sffamily\small

  \foreach \k [evaluate=\k as \n using int(\k+1)] in {0,1,2,3,4}

  \evmark{Ag2}{cXfer}{2}
\end{tikztimingtable}\par
