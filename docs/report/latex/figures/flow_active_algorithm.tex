\begin{tikzpicture}[
		font=\scriptsize,
		term/.style={rectangle, rounded corners=7pt, draw=green!45!black,

		start/.style={term, draw=black!60, fill=blue!8},
		proc/.style={rectangle, draw=black!70, fill=blue!4, align=center,

		busproc/.style={proc, draw=orange!75!black, fill=orange!8},
		resp/.style={proc, text width=3.3cm, draw=violet!65!black, fill=violet!8},
		dec/.style={diamond, aspect=2.35, draw=orange!75!black, fill=orange!10,
				align=center, inner sep=1pt, text width=1.9cm},
		edge/.style={-{Latex[length=1.7mm]}, semithick, draw=black!80},
		join/.style={semithick, draw=black!80},
		err/.style={-{Latex[length=1.7mm]}, semithick, dashed, draw=red!70!black},
		loop/.style={-{Latex[length=1.7mm]}, semithick, dashed, draw=violet!75!black},
		junction/.style={circle, draw=black!75, fill=white, line width=0.8pt,
				minimum size=3.2mm, inner sep=0pt},
		lbl/.style={font=\scriptsize, inner sep=1.4pt, fill=white,
				align=center, text=black!85},
		note/.style={font=\scriptsize\itshape, text=black!65, align=left},
	]
	% Command acquisition and validation.
	\node[start] (start) at (0, 0.00) {\texttt{Idle}};
	\node[proc]  (wait)  at (0,-1.30) {\texttt{WaitForCmd}};
	\node[proc]  (fetchdat) at (0,-2.60) {\texttt{FetchDAT}};
	\node[dec]   (valid) at (0,-4.20) {Descriptor valid?};
	\node[proc]  (waitdat) at (0,-5.85) {\texttt{WaitDAT}};
	\node[dec]   (class) at (0,-7.50) {Command class?};

	% Three execution classes.
	\node[busproc] (ccc) at (-5.0,-9.45)
	{\texttt{I3CBcastHeader}\\$\downarrow$\\\texttt{IssueImmediateCcc}};
	\node[busproc] (private) at (0,-9.45)
	{\texttt{I3CBcastHeader}\\\textit{(optional)}};
	\node[busproc] (daa) at (5.0,-9.45)
	{\texttt{I3CBcastHeader}\\$\downarrow$\\\texttt{IssueDAA}};

	% Private-transfer direction and protocol-specific data handling.
	\node[dec] (dir) at (0,-11.25) {Direction?};
	\node[busproc] (initwrite) at (-2.65,-12.95) {\texttt{InitWrite}};
	\node[busproc] (initread) at (2.65,-12.95) {\texttt{InitRead}};
	\node[busproc] (write) at (-2.65,-14.55)
	{\texttt{FetchTxData}\\$\downarrow$\\\texttt{IssueI3CWrite} / \texttt{IssueI2CWrite}};
	\node[busproc] (read) at (2.65,-14.55)
	{\texttt{IssueI3CRead} / \texttt{IssueI2CRead}};

	% Completion policy and response handling.
	\node[dec]  (cont) at (0,-16.75)
	{\texttt{toc}=0 chained I3C?};
	\node[junction] (finish) at (0,-18.55) {};
	\node[dec]  (needresp) at (0,-20.35) {Response required?};
	\node[resp] (writeresp) at (5.0,-20.35) {\texttt{WriteResp}};
	\node[term] (idle) at (0,-22.35) {\texttt{Idle}};

	% Main downward path.
	\draw[edge] (start) -- (wait);
	\draw[edge] (wait) -- node[lbl,right]{\texttt{cmd\_valid\_i}} (fetchdat);
	\draw[edge] (fetchdat) -- (valid);
	\draw[edge] (valid) -- node[lbl,right]{Yes} (waitdat);
	\draw[edge] (waitdat) -- (class);
	\draw[edge] (class) -- node[lbl,above,sloped]{CCC} (ccc);
	\draw[edge] (class) -- node[lbl,right]{Private} (private);
	\draw[edge] (class) -- node[lbl,above,sloped]{ENTDAA} (daa);
	\draw[edge] (private) -- (dir);
	\draw[edge] (dir) -- node[lbl,above,sloped]{Write} (initwrite);
	\draw[edge] (dir) -- node[lbl,above,sloped]{Read} (initread);
	\draw[edge] (initwrite) -- (write);
	\draw[edge] (initread) -- (read);
	\draw[join] (write.south) -- (-2.65,-15.70) -- (0,-15.70);
	\draw[edge] (read.south) -- (2.65,-15.70) -- (0,-15.70) -- (cont.north);
	\draw[edge] (cont) -- node[lbl,right]{No} (finish);
	\draw[edge] (finish) -- (needresp);
	\draw[edge] (needresp) -- node[lbl,right]{No} (idle);

	% CCC and DAA bypass the chained-private-transfer decision.
	\draw[edge] (ccc.south) -- (-5.0,-18.55)
	-- node[lbl,above]{complete} (finish.west);
	\draw[edge] (daa.south) -- node[lbl,right]{complete} (5.0,-18.55)
	-- (finish.east);

	\draw[err] (needresp.east) -- node[lbl,above]{Yes} (writeresp.west);
	\draw[edge] (writeresp.south) -- node[lbl,right]{\texttt{resp\_queue\_wready\_i}}
	(5.0,-22.35) -- (idle.east);

	% A supported chained I3C command re-enters FetchDAT without releasing the bus.
	\draw[loop] (cont.west) -- node[lbl,above]{Yes} (-7.0,-16.75)
	-- (-7.0,-2.60) -- (fetchdat.west);

	\node[note, anchor=north west, text width=11.8cm] at (-7.0,-23.25)
	{Rectangles: \texttt{flow\_active} states; diamonds: transition conditions;
	dashed paths: response or chained command.};
