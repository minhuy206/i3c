\section{PHY (\texttt{i3c_phy})}
\brief{Purpose: 2FF metastability synchronizer on SCL/SDA inputs + OD/PP output drivers
(active-low open-drain modelling). Interface and bypass/output-select behaviour. Key choice:
minimal PHY, no pad primitives (FPGA IOBUF deferred to future work). $\sim$52 LoC. Spec 01.}

\section{CSR register file + 32-entry DAT (\texttt{csr_register})}
\brief{Purpose: hand-written 32-bit register file + DAT + CMD/TX staging. Interface = the
12-bit/32-bit reg bus. Key choice: hand-written $\sim$300 LoC vs PeakRDL auto-gen
($\sim$14k) --- educational + tiny. Register map at overview level (full bitfields to
Appendix A). Spec 07.}

\tabph{Full CSR map (30 registers + 32-entry DAT)}{csr_register.sv (or defer full table to Appendix A)}

\section{HCI queues (\texttt{hci_queues} / \texttt{sync_fifo})}
\brief{Purpose: four sync FIFOs (CMD 64-bit, TX/RX/RESP 32-bit) built from one generic
\texttt{sync_fifo}. Interface = \texttt{*_valid}/\texttt{*_ready} handshakes. Key choice:
power-of-2 depth enforced by an elaboration-time assertion (``Depth must be a power of 2'')
--- exercised by test FIFO_005. Spec 06.}

\section{\texttt{bus_monitor} (START/STOP/Sr edge detection)}
\brief{Purpose: detect START/STOP/repeated-START and bus-idle from synchronized SCL/SDA using
\texttt{edge_detector} + \texttt{stable_high_detector}. Interface + the +1 capture flop. Spec 02.}

\section{SCL generator (\texttt{scl_generator}, 14-state FSM)}
\brief{Purpose: generate SCL plus START/Sr/STOP/bus-free timing via a single-counter strategy
(area win vs N parallel timers). 14 states (Idle...BusFree). Key choice: DAA restart folded
into \texttt{gen_rstart_i}. Note (future cleanup): \texttt{req_restart_i}/\texttt{ack_o} ports
exist but the RTL routes restart through \texttt{gen_rstart_i} --- a documented cleanup, not a bug.}

\section{TX/RX serializers (\texttt{bus_tx}, \texttt{bus_tx_flow}, \texttt{bus_rx_flow})}
\brief{\texttt{bus_tx_flow} (4-state byte/bit serializer) feeds \texttt{bus_tx} (5-state
per-bit timing engine); \texttt{bus_rx_flow} (4-state) deserializes. Key choice:
\texttt{bus_rx_flow} emits 7 stored bits + live SDA on cycle 8 so the aligned byte is valid on
\texttt{rx_done_o}. Specs 04/05.}

\section{ENTDAA subsystem (\texttt{entdaa_controller} 7-state + \texttt{entdaa_fsm} 8-state)}
\brief{Master-perspective rewrite of the reference's target-side logic. Outer
\texttt{entdaa_controller} (7-state) manages the DAA loop: reads DAT[round], invokes the inner
FSM, increments round until \texttt{no_device}/\texttt{bus_stop_det_i} to Done. Inner
\texttt{entdaa_fsm} (8-state) does the per-device handshake: send 0x7E, shift 64-bit
PID/BCR/DCR, send \{addr, parity\}, return \texttt{addr_valid}; \texttt{bus_stop_det_i} to
NoDev. Spec 08; \texttt{DatDepth}=32.}
